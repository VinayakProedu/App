rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() { return request.auth != null; }
    function isAdmin() { return isSignedIn() && request.auth.token.email == 'vinayakproedu@gmail.com'; }
    function isOwner(uid) { return isSignedIn() && request.auth.uid == uid; }
    function myDoc() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    function myRole() { return myDoc().role; }
    function isPaidStudent() { return isSignedIn() && myRole() == 'student'; }
    function isTeacher() { return isSignedIn() && myRole() == 'teacher'; }
    function isStaff() { return isSignedIn() && (myRole() == 'student' || myRole() == 'teacher'); }
    function assignedSubjects() { return myDoc().assignedSubjects; }

    match /users/{uid} {
      allow create: if isAdmin() || (isOwner(uid) && request.resource.data.role == 'trial');
      allow read: if isOwner(uid) || isAdmin() || isTeacher();   // teachers see student list for their subjects (client filters)
      allow update: if isAdmin() || (isOwner(uid) && request.resource.data.role == 'trial' && request.resource.data.batchCode == '');
      allow delete: if isAdmin();
    }
    match /lectures/{doc} {
      allow read: if isAdmin() || (isSignedIn() && (resource.data.isFree == true || isStaff()));
      allow write: if isAdmin() || isTeacher();
    }
    match /batches/{doc} { allow read: if isSignedIn(); allow write: if isAdmin(); }
    match /announcements/{doc} {
      allow read: if true;
      // Teachers may announce only for subjects assigned to them
      allow create: if isAdmin() || (isTeacher() && request.resource.data.subject in assignedSubjects());
      allow update, delete: if isAdmin();
    }
    match /mockPapers/{doc} {
      allow read: if isAdmin() || (isSignedIn() && (resource.data.isFree == true || isStaff()));
      allow write: if isAdmin() || isTeacher();
    }
    match /mockQuestions/{doc} {
      allow read: if isAdmin() || (isSignedIn() &&
        (get(/databases/$(database)/documents/mockPapers/$(resource.data.paperId)).data.isFree == true || isStaff()));
      allow write: if isAdmin() || isTeacher();
    }
    match /mockAttempts/{doc} {
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow read: if isAdmin() || isTeacher() || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow update, delete: if isAdmin();
    }
    match /studentDoubts/{doc} {
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow read: if isAdmin() || isTeacher() || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow update: if isAdmin() || isTeacher();
      allow delete: if isAdmin();
    }
    match /tests/{doc} {
      allow read: if isAdmin() || (isSignedIn() && (resource.data.isFree == true || isStaff()));
      allow write: if isAdmin() || isTeacher();
    }
    match /questions/{doc} {
      allow read: if isAdmin() || (isSignedIn() &&
        (get(/databases/$(database)/documents/tests/$(resource.data.testId)).data.isFree == true || isStaff()));
      allow write: if isAdmin() || isTeacher();
    }
    match /attempts/{doc} {
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow read: if isAdmin() || isTeacher() || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow update, delete: if isAdmin();
    }
    match /fcmTokens/{doc} { allow create: if isSignedIn(); allow read, update, delete: if false; }
    match /{document=**} { allow read, write: if false; }
  }
}
